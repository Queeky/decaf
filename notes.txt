Set turns: 

SET @count := 0;
UPDATE PLAYER SET PLAY_TURN = @count := @count + 1
WHERE GAME_ID = ? ORDER BY RAND();
UPDATE GAME SET GAME_RUN = 1 WHERE GAME_ID = ?; 

Set game back to GAME_RUN = 0: 

UPDATE GAME SET GAME_TURN = 1, GAME_RUN = 0 WHERE GAME_ID = ?;
